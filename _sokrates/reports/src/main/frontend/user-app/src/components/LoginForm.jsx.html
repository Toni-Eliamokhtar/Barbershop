import { useState } from "react"; export default function LoginForm({ onSubmit }) { const [email, setEmail] = useState(""); const [password, setPassword] = useState(""); const handleSubmit = (e) => { e.preventDefault(); if (onSubmit) onSubmit({ email, password }); }; return ( <form onSubmit={handleSubmit} className="auth-form"> <input type="email" placeholder="Adresse email" value={email} onChange={(e) => setEmail(e.target.value)} required /> <input type="password" placeholder="Mot de passe" value={password} onChange={(e) => setPassword(e.target.value)} required /> <button type="submit" className="auth-btn"> Se connecter </button> </form> ); }
