export async function markBookingPaid(bookingId, paymentInfo = {}) {
  try {
    const { data, error } = await supabase
      .from("bookings")
      .update({ paid: true, payment_info: paymentInfo })
      .eq("id", bookingId)
      .select()
      .single();
    return { data, error };
  } catch (err) {
    return { data: null, error: err };
  }
}
