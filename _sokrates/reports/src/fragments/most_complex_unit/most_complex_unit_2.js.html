export async function createBooking(booking) {
  try {
    const { data, error } = await supabase
      .from("bookings")
      .insert([{ ...booking, paid: false }])
      .select()
      .single();
    return { data, error };
  } catch (err) {
    return { data: null, error: err };
  }
}
